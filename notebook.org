template<typename S>
struct RNG
{
private:
    std::mt19937 gen;
    std::uniform_real_distribution<S> distrib;
public:
    RNG(const int seed):
        gen( seed),
        distrib(S(0), S(1))
    {}

    /** Returns a scalar in [-1, 1] */
    S
    scalar(const S min=S(-1), const S max=S(1))
    {
        return distrib(gen)*(max-min) + min;
    }
    template<uintptr_t SIZE>
    std::array<S, SIZE>
    scalar_array(const S min=S(-1), const S max=S(1))
    {
        std::array<S, SIZE> ar;
        for (S & val: ar)
        {
            val = scalar(min, max);
        }
        return ar;
    }

};

const double EPSILON = 0.00001;

double
sgn(const double x)
{
    return double((x > EPSILON) - (x < -EPSILON));
}


void
test(RNG<double> & rng)
{
    const auto [aa,bb,cc,dd] = rng.scalar_array<4>();
    const double mag = std::sqrt(aa*aa + bb*bb + cc*cc + dd*dd);
    const double a = aa/mag;  const double b = bb/mag;
    const double c = cc/mag;  const double d = dd/mag;
    const double D0 = b*b - 3.0*a*c;
    const double D1 = 2.0*b*b*b - 9.0*a*b*c + 27*a*a*d;
    const double delt2 = D1*D1-4.0*D0*D0*D0;
    const std::complex<double> delt = std::sqrt(std::complex{delt2, 0.0});
    const double s0 = sgn(D0);
    const double s1 = sgn(D1);
    const std::complex<double> gamma_p = std::pow((D1 + delt)/2.0, 1.0/3.0);
    const std::complex<double> gamma_m = std::pow((D1 - delt)/2.0, 1.0/3.0);
    const std::complex<double> gamma =  s1*   std::pow((std::abs(D1)+delt)/2.0, 1.0/3.0);
    const std::complex<double> gamma2 = s0*s1*std::pow(s0*((std::abs(D1)-delt)/2.0), 1.0/3.0);
//        const std::complex<double> gamma = double(sgn(D1))*std::pow((std::abs(D1)+std::pow(std::complex{D1*D1-4.0*D0*D0*D0, 0.0}, 0.5))/2.0,1.0/3.0);


    const auto third = std::polar(1.0, M_PI*2.0/3.0);

    const auto correction = (sgn(D0) < 0.0)? 1.0/third : std::complex<double>{1.0, 0.0};

    const auto correction2 = (sgn(D0) < 0.0)? -2.0 : 1.0;
    const auto gamma_m_corrected = gamma_m*correction;
    const auto gamma_m_corrected2 = gamma_m*correction2;
    
    const double v = 3.0*a;
    const double u = -(b + std::real(gamma + D0/gamma));
    const double u2 = -(b + std::real(gamma_p + gamma_m_corrected));
    const double u3 = -(b + std::real(gamma_p + gamma_m_corrected2));
    const double u4 = -(b + std::real(gamma + gamma2));

        
    std::cout << "COEFFS: " << a << " " << b << " " << c << " " << d << " ";

    const double residual = a*u*u*u + b*u*u*v + c*u*v*v + d*v*v*v;
    const double residual2 = a*u2*u2*u2 + b*u2*u2*v + c*u2*v*v + d*v*v*v;
    const double residual3 = a*u3*u3*u3 + b*u3*u3*v + c*u3*v*v + d*v*v*v;
    const double residual4 = a*u4*u4*u4 + b*u4*u4*v + c*u4*v*v + d*v*v*v;
    std::cout << ", REST: ," << residual << " , " << residual2 << " , " << residual3 << " , " << residual4;
    std::cout << ", Us ," << u << ", " << u2 << ", " << u3 << ", " << u4;
    std::cout << ", SGN DELT: " << sgn(D0);
    std::cout << ", GAMMAS, " << D0/gamma_p << ", " << gamma_m << std::endl;
}
